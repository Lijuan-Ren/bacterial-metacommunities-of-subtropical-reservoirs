library(tidyverse)
library(vegan)
library(broom)
library(broom.mixed)

library(openxlsx)

# -----------------------------
# 1. Load data
# -----------------------------

otu <- read.csv("Bacteria.otu.csv",
                row.names = 1,
                check.names = FALSE)

meta <- read.csv("sample_metadata.csv",
                 check.names = FALSE)

network <- read.csv("SPIEC_EASI_site_network_metrics.csv",
                    check.names = FALSE)

# Required metadata columns:
# SampleID, Reservoir, Season, LifeType, TSI, TrophicState

# Required network columns:
# SampleID, Network_stability, Robustness, Network_complexity
# plus other network properties if needed

# Remove taxonomy column if present
if ("taxa" %in% colnames(otu)) {
  otu_count <- otu[, colnames(otu) != "taxa"]
} else {
  otu_count <- otu
}

# Match samples
shared_samples <- Reduce(
  intersect,
  list(colnames(otu_count), meta$SampleID, network$SampleID)
)

otu_count <- otu_count[, shared_samples]

meta <- meta %>%
  filter(SampleID %in% shared_samples) %>%
  arrange(match(SampleID, shared_samples))

network <- network %>%
  filter(SampleID %in% shared_samples) %>%
  arrange(match(SampleID, shared_samples))

otu_count <- otu_count[, meta$SampleID]

# -----------------------------
# 2. Alpha diversity (richness only)
# -----------------------------

# vegan requires samples as rows
comm <- t(otu_count)

alpha_df <- data.frame(
  SampleID = rownames(comm),
  Richness = specnumber(comm)
)

# Merge richness, metadata, and network metrics
dat <- meta %>%
  left_join(alpha_df, by = "SampleID") %>%
  left_join(network, by = "SampleID")

# -----------------------------
# 3. Define variables
# -----------------------------

# OTU richness is the sole biodiversity predictor
biodiversity_vars <- c("Richness")

network_vars <- c(
  "Nodes",
  "Edges",
  "Average_degree",
  "Density",
  "Clustering_coefficient",
  "Degree_centralization",
  "Average_path_length",
  "Modularity"
)

stability_vars <- c(
  "Network_stability",
  "Robustness",
  "Network_complexity"
)

network_vars <- intersect(network_vars, colnames(dat))
stability_vars <- intersect(stability_vars, colnames(dat))

# Ensure factors are correctly specified
dat <- dat %>%
  mutate(
    Reservoir    = factor(Reservoir),
    Season       = factor(Season),
    LifeType     = factor(LifeType),
    TrophicState = factor(TrophicState)
  )

# -----------------------------
# 4. Global LMM: response ~ TSI
#    (accounts for within-reservoir dependence)
# -----------------------------

run_lmm_tsi <- function(data, response) {

  df <- data %>%
    select(TSI, Season, LifeType, TrophicState,
           Reservoir, all_of(response)) %>%
    drop_na()

  if (nrow(df) < 10 || nlevels(droplevels(df$Reservoir)) < 3) {
    return(NULL)
  }

  fit <- lmerTest::lmer(
    as.formula(paste(
      response,
      "~ TSI * Season * LifeType * TrophicState + (1 | Reservoir)"
    )),
    data = df,
    REML = TRUE
  )

  # Type III Wald F tests with Satterthwaite df
  anova_tab <- as.data.frame(anova(fit, type = 3))
  anova_tab$Term <- rownames(anova_tab)
  anova_tab$Response <- response
  anova_tab$N <- nrow(df)

  anova_tab %>%
    select(Response, Term, N,
           `F value`, `Pr(>F)`, everything()) %>%
    as_tibble()
}

response_vars <- c(
  biodiversity_vars,
  network_vars,
  stability_vars
)

lmm_tsi_results <- bind_rows(
  lapply(response_vars, function(v) {
    run_lmm_tsi(dat, v)
  })
)

write.csv(
  lmm_tsi_results,
  "LMM_TSI_biodiversity_network_stability.csv",
  row.names = FALSE
)

# -----------------------------
# 5. Visualization:
#    TSI vs biodiversity / network metrics
# -----------------------------

plot_tsi_data <- dat %>%
  pivot_longer(
    cols = all_of(response_vars),
    names_to = "Metric",
    values_to = "Value"
  )

p_tsi <- ggplot(
  plot_tsi_data,
  aes(x = TSI, y = Value, color = TrophicState)
) +
  geom_point(size = 2.2, alpha = 0.75) +
  geom_smooth(
    method = "lm",

    linewidth = 0.9
  ) +
  facet_grid(Metric ~ Season + LifeType,
             scales = "free_y") +
  theme_bw() +
  theme(
    strip.text = element_text(face = "bold"),
    panel.grid = element_blank()
  ) +
  labs(
    x = "Trophic state index (TSI)",
    y = "Biodiversity / network metric",
    color = "Trophic state"
  )

ggsave(
  "LMM_TSI_biodiversity_network_stability.pdf",
  p_tsi,
  width = 14,
  height = 18
)

# -----------------------------
# 6. Global LMM: biodiversity-network stability coupling
#    Network property ~ Richness * Season * LifeType *
#                       TrophicState + (1 | Reservoir)
# -----------------------------

run_coupling_lmm <- function(data, response, predictor = "Richness") {

  df <- data %>%
    select(all_of(c(response, predictor)),
           Season, LifeType, TrophicState, Reservoir) %>%
    drop_na()

  if (nrow(df) < 10 || nlevels(droplevels(df$Reservoir)) < 3) {
    return(NULL)
  }

  fit <- lmerTest::lmer(
    as.formula(paste(
      response,
      "~", predictor,
      "* Season * LifeType * TrophicState + (1 | Reservoir)"
    )),
    data = df,
    REML = TRUE
  )

  # Global Type III tests for each fixed effect
  anova_tab <- as.data.frame(anova(fit, type = 3))
  anova_tab$Term <- rownames(anova_tab)
  anova_tab$Response <- response
  anova_tab$Predictor <- predictor
  anova_tab$N <- nrow(df)

  global_tab <- anova_tab %>%
    select(Response, Predictor, Term, N,
           `F value`, `Pr(>F)`, everything()) %>%
    as_tibble()

  # Group-specific slopes from model contrasts (emtrends)
  emm <- emmeans::emtrends(
    fit,
    ~ Season + LifeType + TrophicState,
    var = predictor
  )

  slopes_tab <- as.data.frame(emm) %>%
    as_tibble() %>%
    rename(
      Slope = paste0(predictor, ".trend"),
      SE    = SE,
      df    = df,
      t     = t.ratio,
      P_raw = p.value
    ) %>%
    mutate(
      Response  = response,
      Predictor = predictor,
      # BH-adjust across all group-specific slopes for this model
      P_adj = p.adjust(P_raw, method = "BH"),
      # 95% CI
      CI_low  = Slope - qt(0.975, df) * SE,
      CI_high = Slope + qt(0.975, df) * SE
    ) %>%
    select(Response, Predictor, Season, LifeType, TrophicState,
           Slope, SE, df, CI_low, CI_high, t, P_raw, P_adj)

  list(global = global_tab, slopes = slopes_tab)
}

coupling_out <- lapply(stability_vars, function(resp) {
  run_coupling_lmm(dat, resp, predictor = "Richness")
})

coupling_global <- bind_rows(
  lapply(coupling_out, function(x) if (is.null(x)) NULL else x$global)
)

coupling_slopes <- bind_rows(
  lapply(coupling_out, function(x) if (is.null(x)) NULL else x$slopes)
)

write.csv(
  coupling_global,
  "Biodiversity_stability_coupling_LMM_global.csv",
  row.names = FALSE
)

write.csv(
  coupling_slopes,
  "Biodiversity_stability_coupling_LMM_group_slopes.csv",
  row.names = FALSE
)
