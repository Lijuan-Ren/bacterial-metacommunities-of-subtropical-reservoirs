library(tidyverse)
library(broom)
library(broom.mixed)
library(lme4)
library(lmerTest)
library(emmeans)
library(ggplot2)
library(openxlsx)

# ------------------------------------------------------------
# Input data should contain at least:
# SampleID, Reservoir, Season, LifeType, TrophicState,
# Richness,
# Network_stability, Robustness, Network_complexity
# ------------------------------------------------------------

dat <- read.csv("SPIEC_EASI_site_network_metrics_alpha_diversity.csv",
                check.names = FALSE)

# optional: standardize names
dat <- dat %>%
  rename(
    TrophicState = `Trophic state`
  )

# Ensure factors are correctly specified
dat <- dat %>%
  mutate(
    Reservoir    = factor(Reservoir),
    Season       = factor(Season),
    LifeType     = factor(LifeType),
    TrophicState = factor(TrophicState)
  )

# ------------------------------------------------------------
# 1. Define response and predictor variables
# ------------------------------------------------------------

response_vars <- c(
  "Network_stability",
  "Robustness",
  "Network_complexity"
)

# OTU richness is the sole biodiversity predictor
predictor_var <- "Richness"

# ------------------------------------------------------------
# 2. Global LMM function:
#    Network property ~ Richness * Season * LifeType *
#                       TrophicState + (1 | Reservoir)
# ------------------------------------------------------------

run_biodiv_stability_lmm <- function(data, response, predictor = "Richness") {

  df <- data %>%
    select(
      all_of(c(response, predictor)),
      Season, LifeType, TrophicState, Reservoir
    ) %>%
    drop_na()

  if (nrow(df) < 10 || nlevels(droplevels(df$Reservoir)) < 3) {
    return(NULL)
  }

  fit <- lmerTest::lmer(
    as.formula(paste(
      response,
      "~", predictor,
      "* Season * LifeType * TrophicState + (1 | Reservoir)"
    )),
    data = df,
    REML = TRUE
  )

  # ---- Global Type III tests for each fixed effect ----
  anova_tab <- as.data.frame(anova(fit, type = 3))
  anova_tab$Term <- rownames(anova_tab)
  anova_tab$Response <- response
  anova_tab$Predictor <- predictor
  anova_tab$N <- nrow(df)

  global_tab <- anova_tab %>%
    select(Response, Predictor, Term, N,
           `F value`, `Pr(>F)`, everything()) %>%
    as_tibble()

  # ---- Group-specific slopes from model contrasts ----
  emm <- emmeans::emtrends(
    fit,
    ~ Season + LifeType + TrophicState,
    var = predictor
  )

  slopes_tab <- as.data.frame(emm) %>%
    as_tibble() %>%
    rename(
      Slope  = paste0(predictor, ".trend"),
      SE     = SE,
      df     = df,
      t      = t.ratio,
      P_raw  = p.value
    ) %>%
    mutate(
      Response  = response,
      Predictor = predictor,
      # BH-adjust across all group-specific slopes for this model
      P_adj = p.adjust(P_raw, method = "BH"),
      # 95% CI
      CI_low  = Slope - qt(0.975, df) * SE,
      CI_high = Slope + qt(0.975, df) * SE
    ) %>%
    select(Response, Predictor, Season, LifeType, TrophicState,
           Slope, SE, df, CI_low, CI_high, t, P_raw, P_adj)

  list(global = global_tab, slopes = slopes_tab)
}

# ------------------------------------------------------------
# 3. Run global LMMs for each network property
# ------------------------------------------------------------

lmm_out <- lapply(response_vars, function(resp) {
  run_biodiv_stability_lmm(dat, resp, predictor = predictor_var)
})

lmm_global <- bind_rows(
  lapply(lmm_out, function(x) if (is.null(x)) NULL else x$global)
)

lmm_slopes <- bind_rows(
  lapply(lmm_out, function(x) if (is.null(x)) NULL else x$slopes)
)

# ------------------------------------------------------------
# 4. Save LMM results
# ------------------------------------------------------------

write.csv(
  lmm_global,
  "Biodiversity_stability_coupling_LMM_global.csv",
  row.names = FALSE
)

write.csv(
  lmm_slopes,
  "Biodiversity_stability_coupling_LMM_group_slopes.csv",
  row.names = FALSE
)

# ------------------------------------------------------------
# 5. Plot biodiversity–network stability relationships
#    OTU richness as the sole predictor
# ------------------------------------------------------------

plot_data <- dat %>%
  select(
    SampleID, Season, LifeType, TrophicState,
    Richness, all_of(response_vars)
  ) %>%
  pivot_longer(
    cols = all_of(response_vars),
    names_to = "Network_metric",
    values_to = "Network_value"
  )

p <- ggplot(
  plot_data,
  aes(x = Richness,
      y = Network_value,
      color = TrophicState)
) +
  geom_point(size = 2.2, alpha = 0.75) +
  geom_smooth(
    method = "lm",
    se = TRUE,
    linewidth = 0.9
  ) +
  facet_grid(
    Network_metric ~ Season + LifeType,
    scales = "free"
  ) +
  theme_bw() +
  theme(
    strip.text = element_text(face = "bold"),
    panel.grid = element_blank()
  ) +
  labs(
    x = "OTU richness",
    y = "Network stability / robustness / complexity",
    color = "Trophic state"
  )

ggsave(
  "Biodiversity_stability_coupling_all.pdf",
  p,
  width = 12,
  height = 10
)

# ------------------------------------------------------------
# 6. Plot separately for each season × lifetype
# ------------------------------------------------------------

dir.create("Biodiversity_stability_coupling_plots",
           showWarnings = FALSE)

for (ss in unique(dat$Season)) {
  for (lt in unique(dat$LifeType)) {

    sub_plot <- plot_data %>%
      filter(
        Season == ss,
        LifeType == lt
      )

    p_i <- ggplot(
      sub_plot,
      aes(x = Richness,
          y = Network_value,
          color = TrophicState)
    ) +
      geom_point(size = 2.2, alpha = 0.75) +
      geom_smooth(
        method = "lm",
        se = TRUE,
        linewidth = 0.9
      ) +
      facet_grid(
        Network_metric ~ .,
        scales = "free"
      ) +
      theme_bw() +
      theme(
        strip.text = element_text(face = "bold"),
        panel.grid = element_blank()
      ) +
      labs(
        title = paste(ss, lt),
        x = "OTU richness",
        y = "Network stability / robustness / complexity",
        color = "Trophic state"
      )

    ggsave(
      filename = paste0(
        "Biodiversity_stability_coupling_plots/",
        "Biodiversity_stability_",
        ss, "_", lt, ".pdf"
      ),
      plot = p_i,
      width = 10,
      height = 8
    )
  }
}
