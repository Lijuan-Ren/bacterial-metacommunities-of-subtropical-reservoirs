# -----------------------------
# 2. Statistical analysis of assembly processes
#    among the eight season x lifestyle x trophic-state groups
# -----------------------------

library(tidyverse)
library(ggpubr)
library(rstatix)
library(multcompView)

## Read data
dat <- read.csv(
  "pairwise_TSI_diff_iCAMP_same_trophicalstate.csv",
  check.names = FALSE
)

## Order of trophic-state groups
state_order <- c(
  "WFOM", "WFE", "WPOM", "WPE",
  "DFOM", "DFE", "DPOM", "DPE"
)

## Community assembly processes
process_cols <- c(
  "Heterogeneous.Selection",
  "Homogeneous.Selection",
  "Dispersal.Limitation",
  "Drift.and.Others"
)

## Convert to long format
dat_long <- dat %>%
  pivot_longer(
    cols = all_of(process_cols),
    names_to = "Process",
    values_to = "Contribution"
  ) %>%
  mutate(
    trophicalstate = factor(
      trophicalstate,
      levels = state_order
    )
  )

###############################################################
## Kruskal-Wallis tests
## Compare the relative contribution of each assembly process
## among the eight season x lifestyle x trophic-state groups.
###############################################################

kw_results <- dat_long %>%
  group_by(Process) %>%
  kruskal_test(Contribution ~ trophicalstate)

kw_results

write.csv(
  kw_results,
  "Assembly_process_KruskalWallis.csv",
  row.names = FALSE
)

###############################################################
## Dunn post hoc tests
## Run ONLY when the corresponding Kruskal-Wallis test is
## significant; apply BH adjustment for multiple comparisons.
###############################################################

# Identify processes with a significant Kruskal-Wallis result
sig_processes <- kw_results %>%
  filter(p < 0.05) %>%
  pull(Process)

dunn_results <- dat_long %>%
  filter(Process %in% sig_processes) %>%
  group_by(Process) %>%
  dunn_test(
    Contribution ~ trophicalstate,
    p.adjust.method = "BH"
  )

dunn_results

write.csv(
  dunn_results,
  "Assembly_process_Dunn_posthoc_BH.csv",
  row.names = FALSE
)

###############################################################
## Generate significance letters
## Only for processes with significant Kruskal-Wallis results.
###############################################################

make_letters <- function(df) {

  pw <- dunn_test(
    df,
    Contribution ~ trophicalstate,
    p.adjust.method = "BH"
  )

  pvec <- pw$p.adj
  names(pvec) <- paste(
    pw$group1,
    pw$group2,
    sep = "-"
  )

  letters <- multcompLetters(
    pvec,
    threshold = 0.05
  )$Letters

  data.frame(
    trophicalstate = names(letters),
    Letters = letters
  )
}

if (length(sig_processes) > 0) {

  letters_df <- dat_long %>%
    filter(Process %in% sig_processes) %>%
    group_split(Process) %>%
    purrr::map_dfr(function(x) {

      out <- make_letters(x)
      out$Process <- unique(x$Process)

      ymax <- x %>%
        group_by(trophicalstate) %>%
        summarise(
          ymax = max(Contribution),
          .groups = "drop"
        )

      left_join(out, ymax, by = "trophicalstate")

    })

} else {

  letters_df <- tibble(
    trophicalstate = character(),
    Letters = character(),
    Process = character(),
    ymax = numeric()
  )
}
